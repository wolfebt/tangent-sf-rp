# **99.Schema.BastionLinterAndValidator — Automated BASTION Regression & Math Suite**
## **1. PURPOSE & USAGE**
This module houses the standalone Python verification suite (bastion_validator.py) for the Tangent /game_rules repository. Run it against any new or edited Markdown file or JSON stat block to verify:


## **1. Zero Legacy Terminology Drift: Scans all prose and tables before the DISCREPANCY & RESTORATION LEDGER for prohibited 1d20/d20, DC, BP, Dexterity, Constitution, Awareness, Sense Motive, Linguistics, Animal Handling, Rolling with Hindrance, and book_and_page strings.**
## **2. Deterministic EUFT Math: Computes exact Credit Values (10 * 4^(CR / 5)), Linear Micro-Modifiers (+1 CR = x1.3, +2 = x1.75, +3 = x2.3, +4 = x3.0, -1 = x0.75), 50% Pristine Material costs, and 10% Scrap mass requirements (5x Credit Value).**
## **3. Zero-Stamina-Scaling Character Validation: Confirms that Vitality and Health scale strictly at +5 points per 1 CP from Base 30/30 and that Stamina (STA) applies exclusively to Base Toughness and Stability Points = STA + 5.**


---




## **2. EXECUTABLE PYTHON SCRIPT (bastion_validator.py)**
```python
#!/usr/bin/env python3
"""
BASTION Deterministic Archival Linter & Mathematical Validator
Validates:
## **1. Markdown files against all 12 Mandatory Mechanical Normalization Rules.**
## **2. EUFT Credit Valuation & Linear Micro-Modifier calculations.**
## **3. Operator 150 CP / 60 SP / Zero-Stamina-Scaling Vitality & Health math.**
4. Mecha Scale Multiplier & Garaging UDU conversions.
"""
import os, re, sys, json


PROHIBITED_PATTERNS = [
    ("Rule 1 (1d20/d20/2d20)", re.compile(r"\b(?:1d20|2d20|d20)\b", re.I)),
    ("Rule 2 (Difficulty Class/DC)", re.compile(r"\b(?:Difficulty Class|DCs?)\b")),
    ("Rule 5 (Build Point/BP)", re.compile(r"\b(?:Build Points?|BP)\b")),
    ("Rule 7 (Dexterity)", re.compile(r"\bDexterity\b", re.I)),
    ("Rule 7 (Constitution)", re.compile(r"\bConstitution\b", re.I)),
    ("Rule 11 (Awareness)", re.compile(r"\bAwareness\b")),
    ("Rule 11 (Sense Motive)", re.compile(r"\bSense Motive\b")),
    ("Rule 11 (Linguistics)", re.compile(r"\bLinguistics\b")),
    ("Rule 11 (Animal Handling)", re.compile(r"\bAnimal Handling\b")),
    ("Rule 3 (Rolling with Hindrance)", re.compile(r"Rolling with Hindrance", re.I)),
    ("Directory Topology (book_and_page)", re.compile(r"book_and_page", re.I)),
]


LINEAR_MICRO_MODS = {
    0: 1.0,
    1: 1.3,
    2: 1.75,
    3: 2.3,
    4: 3.0,
    -1: 0.75,
}


def calculate_euft_value(cr: int, flaw_offset: int = 0) -> dict:
    """Calculates canonical EUFT Credit Value, 50% Pristine Material Cost, and 5x Scrap Mass."""
    if flaw_offset == -1 and cr % 5 == 0:
        base_cr = cr
        mult = 0.75
    else:
        base_cr = (cr // 5) * 5
        offset = cr - base_cr
        mult = LINEAR_MICRO_MODS[offset]
    base_val = 10 * (4 ** (base_cr // 5))
    exact_val = round(base_val * mult)
    return {
        "cr": cr,
        "base_5_cr": base_cr,
        "linear_multiplier": mult,
        "credit_value": exact_val,
        "pristine_material_cost_50pct": round(exact_val * 0.5),
        "scrap_resale_10pct": round(exact_val * 0.1),
        "unrefined_scrap_mass_required_5x": exact_val * 5,
    }


def validate_operator_sheet(data: dict) -> list:
    errors = []
    cp = data.get("cp_budget", {})
    if cp.get("personal_hindrance_cp_gained", 0) > 10:
        errors.append("Rule 10 Violation: Personal Hindrance CP exceeds +10 CP cap.")
    if cp.get("archetype_chassis_cp", 80) + cp.get("customization_cp_spent", 0) - cp.get("personal_hindrance_cp_gained", 0) - cp.get("species_disadvantage_cp_refund", 0) != cp.get("total_cp_budget", 150):
        errors.append("Rule 5 Violation: CP budget does not balance.")
    attrs = data.get("attributes", {})
    pools = data.get("pools", {})
    sta = attrs.get("STA", 0)
    if pools.get("base_toughness_from_sta") != sta:
        errors.append(f"Rule 8 Violation: base_toughness_from_sta ({pools.get('base_toughness_from_sta')}) must equal STA ({sta}).")
    if pools.get("stability_points_max") != sta + 5:
        errors.append(f"Rule 8/Mortality Violation: stability_points_max must equal STA + 5 ({sta + 5}).")
    if pools.get("chassis_biology") == "Biological":
        added_pts = (pools.get("vitality_max", 30) - 30) + (pools.get("health_max", 30) - 30)
        expected_cp = added_pts / 5.0
        if expected_cp != pools.get("cp_spent_on_pools", 0):
            errors.append(f"Rule 8 Violation: Vitality/Health increase ({added_pts} pts) must equal 5 * cp_spent_on_pools ({pools.get('cp_spent_on_pools') * 5}).")
    return errors


def audit_markdown_file(filepath: str) -> list:
    with open(filepath, "r", encoding="utf-8") as f:
        txt = f.read()
    body = re.split(r"^#{2,3}\s+DISCREPANCY & RESTORATION LEDGER", txt, maxsplit=1, flags=re.M)[0]
    # Also skip code/table documentation blocks that explicitly list prohibited terms as examples
    body_clean = re.sub(r"Never use `[^`]+`", "", body)
    body_clean = re.sub(r"PROHIBITED_PATTERNS = \[.*?\]", "", body_clean, flags=re.S)
    findings = []
    for label, pat in PROHIBITED_PATTERNS:
        matches = pat.findall(body_clean)
        if matches:
            findings.append(f"{filepath}: {label} -> {matches}")
    return findings


if __name__ == "__main__":
    target = sys.argv[1] if len(sys.argv) > 1 else "."
    all_findings = []
    md_count = 0
    for root, _, files in os.walk(target):
        for fn in sorted(files):
            if fn.endswith(".md"):
                md_count += 1
                all_findings.extend(audit_markdown_file(os.path.join(root, fn)))
    print(f"BASTION Audit Complete: {md_count} Markdown files scanned, {len(all_findings)} violations found.")
    for item in all_findings:
        print("  [VIOLATION]", item)
    sys.exit(1 if all_findings else 0)

```
